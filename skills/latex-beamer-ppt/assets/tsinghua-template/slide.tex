\section{拓扑空间上的层}

\begin{frame}
\begin{block}{Definition 1.1.}
设 $X$ 为拓扑空间，$\mathcal{C}$ 为一个范畴。拓扑空间 $X$ 上取值于范畴 $\mathcal{C}$ 的\textbf{预层} $\mathcal{F}$ 是一个反变函子
\[
\mathcal{F}: \mathbf{Top}(X) \to \mathcal{C}.
\]
记 $\mathrm{res}^V_U: \mathcal{F}(V) \to \mathcal{F}(U)$ 为限制态射，它是包含映射 $U \hookrightarrow V$ 在函子 $\mathcal{F}$ 下的像。

若预层 $\mathcal{F}$ 满足以下\textbf{局部到整体公理}，则称 $\mathcal{F}$ 为\textbf{层}：

设 $U$ 是 $X$ 的开集，$\{U_i\}$ 是 $U$ 的一个开覆盖。任取截面 $f_i \in \mathcal{F}(U_i)$，如果对所有 $i,j$ 都满足相容条件
\[
\mathrm{res}_{U_i,U_{ij}} f_i = \mathrm{res}_{U_j,U_{ij}} f_j,
\]
那么存在唯一的截面 $f \in \mathcal{F}(U)$，使得对所有 $i$ 都有
\[
\mathrm{res}_{U,U_i} f = f_i.
\]
\end{block}

\end{frame}

\begin{frame}
\begin{block}{Definition 1.2.}

设 $X$ 为拓扑空间，$\mathcal{F},\mathcal{G}$ 是 $X$ 上取值于范畴 $\mathcal{C}$ 的两个层。
从 $\mathcal{F}$ 到 $\mathcal{G}$ 的\textbf{层态射}是其底预层之间的一个自然变换 $\eta: \mathcal{F} \to \mathcal{G}$；
换言之，层态射由一族范畴 $\mathcal{C}$ 中的态射
$$\eta_U: \mathcal{F}(U) \to \mathcal{G}(U)$$
构成，且对每个开集包含 $U \hookrightarrow V$，下述图表交换：

拓扑空间 $X$ 上取值于具体范畴 $\mathcal{C}$ 的全体层构成一个范畴，而且是满足层公理的、取值于 $\mathcal{C}$ 的预层范畴的满子范畴。
我们用 $\mathbf{Ab}(X)$表示 $X$ 上的交换群层范畴，用$\mathbf{Sh}(X)$表示集合层范畴。
有时我们也将 $\mathbf{Ab}(X)$ 中的对象称为交换层。
\end{block}

\end{frame}


\begin{frame}


\begin{block}{Definition 1.3.}
 设$f:X\to Y$为拓扑空间之间的连续映射，定义$f$的\textbf{前推函子}为映射：
\[
f_{*}: \mathbf{Ab}(X) \to \mathbf{Ab}(Y),
\]
其将交换群层$\mathcal{F}$映为$f_{*}\mathcal{F}$，其中$f_{*}\mathcal{F}$对任意$V\in\mathbf{Top}(Y)$的赋值为
\[
f_{*}\mathcal{F}(V) = \mathcal{F}\bigl(f^{-1}(V)\bigr),
\]
$f_{*}\mathcal{F}$关于开集包含的限制行为由$\mathcal{F}$自然继承。称$f_{*}\mathcal{F}\in\mathbf{Ab}(Y)$为$\mathcal{F}$在$f$下的\textbf{前推层}。
\end{block}

\end{frame}

\begin{frame}
\begin{block}{Proposition 1.4.}设$\mathcal{F}\in\mathbf{Ab}(X)$为拓扑空间$X$上的交换群层，则按上述定义得到的$f$下的前推层$f_{*}\mathcal{F}$是$Y$上的交换群层，且该赋值诱导了层范畴之间的函子$f_{*}: \mathbf{Ab}(X) \to \mathbf{Ab}(Y)$。

\end{block}
\end{frame}


\begin{frame}
\begin{block}{Example 1.5.}
(1)设$A$为取定的交换群，$x$为拓扑空间$X$中的一点。通过下述赋值定义摩天大楼层(skyscraper sheaf)$A^x \in \mathbf{Ab}(X)$：
\[
A^x: U \mapsto
\begin{cases}
A, & x \in U, \\
0, & \text{其他情形},
\end{cases}
\]
其限制态射$\mathrm{res}_{V,U}$的取法为：若$x \in U \cap V$，则$\mathrm{res}_{V,U}$为交换群$A$上的恒等态射$\mathrm{id}_A$；其余情形下，为交换群范畴$\mathbf{Ab}$中唯一的零态射。

更一般地，记$j: \{x\} \hookrightarrow X$为自然包含映射，记$A^x \in \mathbf{Ab}(\{x\})$为单点集$\{x\}$上的摩天大楼层（此时$A^x$将$\{x\}$映为$A$，将空集映为$0$）。

\end{block}
\end{frame}
\begin{frame}
我们可取每个$A^x$在包含映射下的前推层$j_*A^x$，并进一步考虑$X$上的直积层$\mathcal{F}:=\prod_{x\in X}(j_*A^x)$，其赋值规则为：
\[
\mathcal{F}(U) = \prod_{x\in X} (j_*A^x)(U),
\]
其中直积在交换群范畴$\mathbf{Ab}$中取通常的直积，限制态射按分量方式定义。由前推层的定义可知$(j_*A^x)(U)=A^x(j^{-1}(U))$，因此上述赋值可简化为：
\[
\mathcal{F}: U \mapsto \prod_{x\in U} A.
\]

\end{frame}

\begin{frame}
\begin{block}{Definition 1.5.}
(2)设$A$为取定的交换群，我们可考虑交换群的常预层$\underline{A}$，该预层将拓扑空间$X$中所有非空开集均映为$A$；对于开集的包含映射$U \hookrightarrow V$，若$U$非空则将其映为恒等态射$\mathrm{id}_A$，若$U$为空集则映为零态射。

易知常预层$\underline{A}$未必是层：不妨设交换群$A$中至少含两个元素，考虑离散两点空间$\{p,q\}=\{p\}\cup\{q\}$上的常预层$\underline{A}$，其在开集$\{p\}$和$\{q\}$上的局部截面分别取$A$中不同的元素$a,b$，此时不存在满足粘合条件的整体截面，故该常预层不满足层的公理。


\end{block}
\end{frame}

由右侧方块的交换性可得
$$\varphi_U \circ \mathrm{res}_{V,U}^{\mathcal{F}} \circ l_V = 0,$$




\end{block}

\end{block}

\section{茎与层化}
\begin{frame}
\begin{block}{Proposition 2.1.}
设 $\mathcal{F}$ 是拓扑空间 $X$ 上的预层。对任意点 $x \in X$，定义 $\mathcal{F}$ 在点 $x$ 处的茎 $\mathcal{F}_x$ 为下述正向极限所给出的交换群：
\[
\mathcal{F}_x := \varinjlim_{\substack{V \ni x,\, V \in \mathbf{Top}(X)}} \mathcal{F}(V). \tag{4.2}
\]

因此，$\mathcal{F}$ 在点 $x$ 处的茎配备了一族相容映射
\[
\bigl(\eta_U: \mathcal{F}(U) \to \mathcal{F}_x\bigr)_{U \in \mathcal{N}_x},
\]
其中 $\mathcal{N}_x = \{U \subset X \text{ 开集} \mid U \ni x\}$ 是点 $x$ 在 $X$ 中的邻域系。相容性是指：对 $\mathcal{N}_x$ 中的任意包含关系 $U \subset V$，对应的三角图
是交换的。


\end{block}
\end{frame}

\begin{frame}
\begin{block}{Definition 2.2.}
给定层态射 $\varphi: \mathcal{F} \to \mathcal{G}$，对任意 $x \in X$，我们自然在茎的层面上
得到诱导的群同态 $\varphi_x: \mathcal{F}_x \to \mathcal{G}_x$。
只需取代表元 $(f, U)$，并定义
\[
\varphi_x\bigl[(f, U)\bigr] := \bigl(\varphi_U(f)\bigr)_x.
\]

$\varphi_x$ 的良定义性可通过对上述图表应用自然性条件直接验证。
即若对某个包含 $x$ 的开集 $W$，有
$\operatorname{res}_{U,W}^{\mathcal{F}} f = \operatorname{res}_{V,W}^{\mathcal{F}} g$，
从而 $(f, U) \sim (g, V)$，则由 $\varphi$ 的自然性可得
\[
\operatorname{res}_{U,W}^{\mathcal{G}}\bigl(\varphi_U f\bigr)
= \varphi_W\bigl(\operatorname{res}_{U,W}^{\mathcal{F}} f\bigr)
= \varphi_W\bigl(\operatorname{res}_{V,W}^{\mathcal{F}} g\bigr)
= \operatorname{res}_{V,W}^{\mathcal{G}}\bigl(\varphi_V g\bigr),
\]
因此在 $\mathcal{G}_x$ 中有 $(\varphi_U(f), U) \sim (\varphi_V(g), V)$。

这一对应作为映射
$(-)_x: \mathbf{Ab}(X) \to \mathbf{Ab}$
显然具有函子性。
\end{block}
\end{frame}


\begin{frame}
\begin{block}{Lemma 2.3}
设 $\varphi: \mathcal{F} \to \mathcal{G}$，$\psi: \mathcal{F} \to \mathcal{G}$ 为层态射，则
$\varphi = \psi$ 当且仅当对所有 $x \in X$ 均有 $\varphi_x = \psi_x$。
\end{block}
\begin{block}{Lemma 2.4}
设 $\varphi: \mathcal{F} \to \mathcal{G}$ 为层态射，则对所有 $x \in X$ 有
\[
(\ker \varphi)_{x} = \ker(\varphi_{x}),
\]
其中我们将 $(\ker \varphi)_{x}$ 经由典范嵌入 $l_{x}: (\ker \varphi)_{x} \hookrightarrow \mathcal{F}_{x}$ 视为 $\mathcal{F}_{x}$ 的子群。

\end{block}
\end{frame}

\begin{frame}
\begin{block}{Construction 2.5}

设 $\mathcal{F}$ 是 $X$ 上的预层。定义 $\mathcal{F}$ 的层化函子
$\mathcal{F}^+: \mathrm{Top}(X)^\mathrm{op} \to \mathbf{Ab}$，其定义为：
\[
\mathcal{F}^+: U \longmapsto \left\{
\begin{array}{l}
\forall x \in U,\ f(x) \in \mathcal{F}_x,\ \text{且} \\
\exists \text{ 开集 } W \ni x,\ \exists g \in \mathcal{F}(W),\ \text{使得} \\
f(y)=g_y \in \mathcal{F}_y,\ \forall y \in W
\end{array}
\right\} \tag{4.9}
\]
\end{block}
\end{frame}

\begin{frame}

换言之，$U$ 上的 $\mathcal{F}^+$ 局部截面是这样一类函数：它们把 $U$ 映入 $X$ 上各茎的无交并，与茎结构相容，并且在茎的层面上局部等同于 $\mathcal{F}$ 的（可能不同的）局部截面。

对包含关系 $U \subset V$，以自然方式定义 $\mathcal{F}^+$ 的限制映射（即直接作为函数的限制）：
\[
\operatorname{res}_{V,U}^{\mathcal{F}^+}f := f|_U,
\]
从而得到 $X$ 上一个合法的预层。进一步不难验证 $\mathcal{F}^+$ 满足层公理，因此实际上有 $\mathcal{F}^+ \in \mathbf{Ab}(X)$。

具体而言，任取 $U\in\mathrm{Top}(X)$ 的一个开覆盖 $\{U_i\}$ 以及相容截面族 $f_i\in\mathcal{F}^+(U_i)$，只需对 $x\in U_i$ 定义 $f(x)=f_i(x)$，即可得到 $f\in\mathcal{F}^+(U)$。这样给出的 $f$ 是良定义的，并且由构造可知，它是唯一满足对所有 $i$ 有
\[
\operatorname{res}_{U,U_i}^{\mathcal{F}^+}f = f_i
\]
的截面。
\end{frame}
\begin{frame}
\begin{block}{Lemma 2.6}
设 $\mathcal{F}$ 是 $X$ 上的预层，$\mathcal{F}^+$ 为其层化，则对所有 $x \in X$，有
\[
\mathcal{F}_x \cong \mathcal{F}_x^+.
\]
\end{block}
\begin{block}{Lemma 2.7}
设 $\mathcal{F}$ 是拓扑空间 $X$ 上的预层，$\mathcal{F}^+$ 是上文所定义的 $\mathcal{F}$ 的层化。
则存在一个预层态射 $\theta: \mathcal{F}\to\mathcal{F}^+$，并且配对

$\psi: \mathcal{F}\to\mathcal{G}$ 的层 $\mathcal{G}$，
存在唯一的层态射 $\varphi: \mathcal{F}^+\to\mathcal{G}$ 使得下图三角形交换：
\end{block}

\end{frame}

\begin{block}{Construction2.8}
我们现在给出由拓扑空间映射 $f$ 诱导的正向投射函子
$f_{*}: \mathbf{Ab}(X) \to \mathbf{Ab}(Y)$ 的对偶构造。

设 $F\colon \mathcal{A}\to\mathcal{B}$ 为阿贝尔范畴之间的协变、可加、左正合函子，并且假定 $\mathcal{A}$ 具有足够多的内射对象。
若对象 $K\in\mathcal{A}$ 满足对所有 $i>0$ 都有 $R^iF(K)=0$，则称 $K$ 为 \textbf{$F$-零调对象}。

\end{block}

\begin{block}{Proposition 4.9}
设 $F:\mathcal{A}\to\mathcal{B}$ 如上所述，$A$ 是 $\mathcal{A}$ 中的对象，且
\[
0 \to A \stackrel{\epsilon}{\to} K^{0} \to K^{1} \to \cdots
\]
是 $\mathcal{A}$ 中的正合列，满足对所有 $n$，$K^{n}$ 都是 $F$-零调的。则对每个 $i\ge0$，都有自然同构
\[
H^{i}(F(K^{\cdot})) \cong R^{i}F(A).
\]
\end{block}
\end{frame}

\section{层上同调}

\begin{frame}
\begin{block}{Theorem 5.1}
设 $X$ 为拓扑空间，则交换群层范畴 $\mathbf{Ab}(X)$ 具有足够多的内射对象。

\end{block}
\end{frame}

\begin{frame}
现在，通过下式定义态射 $\epsilon: \mathcal{F} \to \mathcal{I}$：
\[
\epsilon_U: s\in\mathcal{F}(U) \longmapsto \prod_{x\in U}\iota_x(s_x).
\]
这是一个良定的层态射。事实上，根据定义，茎通过截面在该点的芽来识别元素，因此对包含关系 $U\hookrightarrow V$ 以及截面 $s\in\mathcal{F}(V)$，对每个 $x\in U$ 都有
\[
s_x = \big(\operatorname{res}_{V,U}^{\mathcal{F}}s\big)_x,
\]
由此可得 $\epsilon$ 与限制映射相容。

此外，$\epsilon$ 是一个单态射：若 $\epsilon_U(s)=0$，则对所有 $x\in U$ 有 $\iota_x(s_x)=0$。由于每个 $\iota_x$ 都是单同态，便有 $s_x=0$ 对所有 $x\in U$ 成立，再由粘合的唯一性可知 $s=0$。
\end{frame}

\begin{frame}
\begin{block}{Definition 5.2}
设 $X$ 为拓扑空间，$\mathcal{F}$ 是 $X$ 上的交换群层。
定义以 $\mathcal{F}$ 为系数的空间 $X$ 的第 $i$ 个层上同调群，是将整体截面函子 $\Gamma(X,-)$ 的第 $i$ 个右导出函子作用在 $\mathcal{F}$ 上得到的群：
\[
H^i(X,\mathcal{F}):=R^i\big(\Gamma(X,-)\big)(\mathcal{F}).\tag{7.5}
\]
\end{block}

一般的层上同调并不一定满足同伦不变性。不过，同伦不变性对某一类层是成立的。
称层 $\mathcal{F}\in\mathbf{Ab}(X)$ 为局部常值层，如果对每个点 $x\in X$，都存在邻域 $U\ni x$ 使得限制层 $\mathcal{F}|_U$ 同构于一个常值层。
\end{frame}
